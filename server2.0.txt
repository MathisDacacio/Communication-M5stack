void handleApi()
{
  if (server.method() == HTTP_POST)
  {
    String body = server.arg("plain");
    StaticJsonDocument<512> doc;
    DeserializationError error = deserializeJson(doc, body);

    if (error)
    {
      server.send(400, "application/json", "{\"status\":\"error\",\"message\":\"Invalid JSON\"}");
      return;
    }

    const char *id = doc["id"];
    bool found = false;

    for (auto &existingDoc : allApiData)
    {
      if (existingDoc["id"] == id)
      {
        existingDoc = doc;
        found = true;
        break;
      }
    }

    if (!found)
    {
      allApiData.push_back(doc);
    }

    server.send(200, "application/json", "{\"status\":\"success\",\"message\":\"Data received and stored\"}");
  }
  else
  {
    server.send(405, "application/json", "{\"status\":\"error\",\"message\":\"Method not allowed\"}");
  }
}

void handleData()
{
  String jsonData = "{\"status\":\"success\",\"data\":[";

  for (size_t i = 0; i < allApiData.size(); ++i)
  {
    String data;
    serializeJson(allApiData[i], data);
    jsonData += data;
    if (i < allApiData.size() - 1)
    {
      jsonData += ",";
    }
  }

  jsonData += "]}";

  server.send(200, "application/json", jsonData);

  // Parse the JSON data
  DynamicJsonDocument doc(1024);
  deserializeJson(doc, jsonData);
}

void handleApiWeather()
{
  HTTPClient http;
  String weatherApiUrl = "https://api.openweathermap.org/data/2.5/weather?q=Lille&units=metric&lang=fr&APPID=dff65ece7cd56e5609a961090c1815a5";

  http.begin(weatherApiUrl);
  int httpResponseCode = http.GET();

  if (httpResponseCode > 0)
  {
    String payload = http.getString();
    StaticJsonDocument<1024> doc;
    DeserializationError error = deserializeJson(doc, payload);

    if (!error)
    {
      server.send(200, "application/json", payload);
    }
    else
    {
      server.send(500, "application/json", "{\"status\":\"error\",\"message\":\"Invalid JSON from API\"}");
    }
  }
  else
  {
    String error = "{\"status\":\"error\",\"message\":\"Unable to fetch weather data\"}";
    server.send(500, "application/json", error);
  }

  http.end();
}

void updateDisplay()
{
  if (!allApiData.empty())
  {
    static String lastLightValue = "";
    static String lastLightState = "";
    static String lastLightUnits = "";
    static String lastPresenceStatus = "";
    static String lastDoorStatus = "";
    static String lastTempValue = "";
    static String lastTempUnits = "";

    for (auto &doc : allApiData)
    {
      String id = doc["id"].as<String>();

      if (id == "light")
      {
        String currentLightValue = doc["value"].as<String>();
        String currentLightState = doc["status"].as<String>();
        String currentLightUnits = doc["units"].as<String>();

        if (currentLightState == "off")
        {
          if (currentLightState != lastLightState)
          {
            M5.Lcd.setCursor(10, 30);
            M5.Lcd.fillRect(10, 30, 300, 20, BLACK);
            M5.Lcd.print("Light: ");
            M5.Lcd.print("Lumiere eteinte");
            lastLightState = currentLightState;
          }
        }
        else
        {
          if (currentLightValue != lastLightValue || currentLightUnits != lastLightUnits)
          {
            M5.Lcd.setCursor(10, 30);
            M5.Lcd.fillRect(10, 30, 300, 20, BLACK);
            M5.Lcd.print("Light: ");
            M5.Lcd.print(currentLightValue);
            M5.Lcd.print(" ");
            M5.Lcd.print(currentLightUnits);
            lastLightValue = currentLightValue;
            lastLightUnits = currentLightUnits;
          }
          lastLightState = currentLightState;
        }
      }
      else if (id == "MotionSensor")
      {
        String currentPresenceStatus = doc["status"].as<String>();

        if (currentPresenceStatus != lastPresenceStatus)
        {
          M5.Lcd.setCursor(10, 50);
          M5.Lcd.fillRect(10, 50, 300, 20, BLACK);
          M5.Lcd.print("Presence: ");
          M5.Lcd.print(currentPresenceStatus);
          lastPresenceStatus = currentPresenceStatus;
        }
      }
      else if (id == "door")
      {
        String currentDoorStatus = doc["status"].as<String>();

        if (currentDoorStatus != lastDoorStatus)
        {
          M5.Lcd.setCursor(10, 70);
          M5.Lcd.fillRect(10, 70, 300, 20, BLACK);
          M5.Lcd.print("Door: ");
          M5.Lcd.print(currentDoorStatus);
          lastDoorStatus = currentDoorStatus;
        }
      }
      else if (id == "temp")
      {
        String currentTempValue = doc["value"].as<String>();
        String currentTempUnits = doc["units"].as<String>();

        if (currentTempValue != lastTempValue || currentTempUnits != lastTempUnits)
        {
          M5.Lcd.setCursor(10, 90);
          M5.Lcd.fillRect(10, 90, 300, 20, BLACK);
          M5.Lcd.print("Temperature: ");
          M5.Lcd.print(currentTempValue);
          M5.Lcd.print(" ");
          M5.Lcd.print(currentTempUnits);
          lastTempValue = currentTempValue;
          lastTempUnits = currentTempUnits;
        }
      }
    }
  }
}

void setup()
{
  M5.begin();
  M5.Lcd.setTextSize(2);
  M5.Lcd.setCursor(10, 10);
  M5.Lcd.print("Connexion a WiFi1...");
  WiFi.mode(WIFI_AP_STA);
  WiFi.begin(wifi_ssid, wifi_password);
  while (WiFi.status() != WL_CONNECTED)
  {
    delay(500);
    M5.Lcd.print(".");
  }
  M5.Lcd.fillScreen(BLACK);
  M5.Lcd.setCursor(10, 10);
  M5.Lcd.print("Connecté à WiFi1");
  M5.Lcd.setCursor(10, 30);
  M5.Lcd.print("IP STA: ");
  M5.Lcd.print(WiFi.localIP());
  // Configurer le point d'accès (mode AP)
  WiFi.softAP(ap_ssid, ap_password);
  M5.Lcd.setCursor(10, 50);
  M5.Lcd.print("Hotspot: Groupe1");
  M5.Lcd.setCursor(10, 70);
  M5.Lcd.print("IP AP: ");
  M5.Lcd.print(WiFi.softAPIP());
  // Configurer les routes
  server.on("/", handleRoot);
  server.on("/api", HTTP_POST, handleApi);
  server.on("/data", HTTP_GET, handleData);
  server.on("/api-weather", HTTP_GET, handleApiWeather);
  server.begin();
  M5.Lcd.setCursor(10, 90);
  M5.Lcd.print("Serveur HTTP actif");
  M5.Lcd.clear();
}

void loop()
{
  server.handleClient();
  int numClients = WiFi.softAPgetStationNum();

  if (numClients != previousNumClients)
  {
    previousNumClients = numClients;
    M5.Lcd.setCursor(10, 10);
    M5.Lcd.fillRect(10, 10, 300, 20, BLACK);
    M5.Lcd.print(numClients);
    M5.Lcd.print(" client(s) AP");
  }

  // Call the function to update the display
  updateDisplay();
}
